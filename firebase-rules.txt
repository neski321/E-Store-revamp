service cloud.firestore {
  match /databases/{database}/documents {
    // Users collection
    match /users/{userId} {
      allow read, write, delete: if request.auth.uid == userId || 
                                   get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';

      // Orders subcollection for each user
      match /orders/{orderId} {
        allow read, write, delete: if request.auth.uid == userId;
      }

      // Favorites subcollection for each user
      match /favorites/{favoriteId} {
        allow read, write, delete: if request.auth.uid == userId;
      }

      // Checkout subcollection for each user
      match /checkout/{checkoutId} {
        allow read, write, delete: if request.auth.uid == userId;
      }
    }

    // Contact Messages Collection
    match /contactMessages/{userId}/messages/{messageId} {
      allow read, write, delete: if request.auth.uid == userId;
    }

    // Admin Requests Collection - for admin access management
    match /adminRequests/{requestId} {
      // Users can create and read their own requests
      allow create: if request.auth.uid == request.resource.data.userId;
      allow read: if request.auth.uid == resource.data.userId || request.auth != null;
      
      // Only admins can update (approve/reject/revoke) requests
      allow update: if request.auth != null;
      
      // Users can delete their own requests, admins can delete any
      allow delete: if request.auth.uid == resource.data.userId || request.auth != null;
    }
  }
}
